# Code patches for Ocarina of Time 1.1.1 main (build id e0f3ccfaf50065ae4ea01eb5aeb13b1a45da3348), recomp/patches.py.
# Ported from Fritchii's built-in patches (src/core/file_sys/patch_manager.cpp, branch oot-gather-fix, GPL-3.0-or-later),
# whose module offsets are these. Each set is switched at start by its key: 0 off, 1 on. In totk.ini the key goes
# without TOTK_ (OOT_MAX_RESOLUTION = 1), as for every other setting there.
# Both are off by default. Max Resolution turns the screen black once gameplay starts here (the runtime does not handle
# the full-size scene yet). Occlusion Off removes the vegetation for good: this runtime's culling depends on the queries.

set maxres    TOTK_OOT_MAX_RESOLUTION 0
set occlusion TOTK_OOT_OCCLUSION_OFF  0

# Max Resolution: the game takes its "dynamic resolution disabled" path, so the 3D renders at full scale instead of
# following the GPU-time controller down to its 480p floor. Fritchii branches unconditionally over the flag test
# (main+0x805bd0 tbz w8, #0 -> b 0x80640c); a lifted site cannot branch, so the flag load before it reads zero instead,
# which makes the tbz always taken. The target's first instruction (ldr w8, [x19, ...]) overwrites w8.
site maxres main 805bcc b9409508 2a1f03e8  # ldr w8, [x8, #0x94] -> mov w8, wzr

# Occlusion Off (cLohan's community patch, zortos293/Lichii#641): the game creates no z-prepass occlusion queries and no
# samples-passed queries, so it issues no occlusion reports or conditional renders; every candidate object is drawn.
site occlusion main 11ace68 39400008 2a1f03e8  # ldrb w8, [x0] -> mov w8, wzr
site occlusion main 11acec8 39400008 2a1f03e8
site occlusion main 11b828c 39400008 2a1f03e8
site occlusion main 11b82a4 39400008 2a1f03e8
site occlusion main 11b82dc 39400008 2a1f03e8
site occlusion main 11b82f4 39400008 2a1f03e8

# VR camera: the renderer's set-view function (main+0x33a410, view index w1) gets the 3x4 view matrix at x2 and again at
# x4 (its inverse and culling planes come from x4). On entry the host function at page+0xfb0 (runtime/vr_camera.c) points
# both at a turned copy. x16 keeps x30, the caller, for it.
set vrcam TOTK_VR_CAMERA 0
site vrcam main 33a428 b940100a b940100a aa1e03f0 952d9ae2  # ldr w10, [x0, #0x10]; mov x16, x30; bl page+0xfb0
# The camera controller's update (main+0x4c48c0, x0 the controller: float eye at +0xd8, float look-at point at +0x100)
# tells the host function at page+0xfa0 (runtime/vr_camera.c) where it is. First person looks along eye to look-at.
site vrcam main 4c48f4 aa0003f3 aa0003f3 aa1e03f0 952771ab  # mov x19, x0; mov x16, x30; bl page+0xfa0
# The camera target's update (main+0x32a4c40) resolves the actor the camera follows (Link) into x0 at main+0x32a4de4; the host
# function at page+0xf90 (runtime/vr_camera.c) keeps it. Its position is at +0x31c.
site vrcam main 32a4de4 aa0003f5 aa0003f5 aa1e03f0 946ff06b  # mov x21, x0; mov x16, x30; bl page+0xf90
# A skeleton's skinning matrices (main+0x14e0200, x0 the skeleton: world matrices at +0x20, 0x40 per bone) are made from its
# world matrices after the physics ran. The host function at page+0xf80 (runtime/vr_camera.c) shrinks Link's head there first.
site vrcam main 14e0220 aa0003f3 aa0003f3 aa1e03f0 94e70358  # mov x19, x0; mov x16, x30; bl page+0xf80
# A cutscene camera track's update (main+0xe7a220, it reads CameraPos / CameraAt / CameraUp) tells the host function at
# page+0xf70 (runtime/vr_camera.c) that a cutscene camera runs. It runs only on area entry, not in talks (docs/vr-port-notes.md).
site vrcam main e7a248 aa0003f3 aa0003f3 aa1e03f0 95009b4a  # mov x19, x0; mov x16, x30; bl page+0xf70
